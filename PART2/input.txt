empty() { "" }

twice(x) { x + x }

rev(x) { reverse x }

palindrome(s) {
  if (s = reverse s) "yes" else "no"
}

endsWithIng(s) {
  if ("ing" suffix s) "ing-word" else "other"
}

classify(a, b) {
  if (a = b)
    if ("x" prefix a) "same-x" else "same"
  else
    if (a + b = b + a)
      "commute"
    else
      if ("ab" suffix a + b) "ends-ab" else "diff"
}

decorate(x) {
  "[" + (if (x = "") "empty" else x) + "]"
}

pick(c, a, b) {
  if (c = "go") a else b + "!"
}

mix(x) {
  "<" + if ("a" prefix x) "A" else "N" + ">"
}

deep(a) {
  reverse reverse reverse a + "|" + reverse (a + "!")
}

greet(n) {
  "Γεια σου, " + n + "! \"quoted\"\ttab"
}

palindrome("level")
palindrome("abc")
palindrome("")
endsWithIng("testing")
endsWithIng("ing")
endsWithIng("in")
classify("x1", "x1")
classify("a", "a")
classify("", "")
classify("a", "b")
classify("ac", "bd")
classify("aa", "a")
decorate("")
decorate("hi")
pick("go", "A", "B")
pick("stop", "A", "B")
mix("abc")
mix("zzz")
deep("ab")
twice(twice("ab"))
rev(twice(rev("ab")))
greet("κόσμε")
twice(if ("a" prefix "ab") "yes" else "no")
empty() + "end"
